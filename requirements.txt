# Core dependencies for qdvc-offgrid-buddies.
#
# IMPORTANT: pin llama-cpp-python to ONE version and use the SAME version on
# every laptop. A save-point built with one version may not restore correctly
# under another, because the serialized KV-cache state format is version-
# sensitive. Replace <x.y.z> with the version you standardise on, e.g.:
#     llama-cpp-python==0.3.2
llama-cpp-python

PyYAML>=6.0

# The `chat` command's text UI (arrow-key buddy picker, streaming styled chat).
# Pin one version across all laptops for consistent rendering.
textual

# Cross-platform clipboard for the chat "copy" buttons. On a headless / off-grid
# Linux laptop, pyperclip needs a backend: install xclip or xsel (X11), or
# wl-clipboard (Wayland). Without one, copy shows a helpful message instead of
# copying.
pyperclip

# Optional but recommended: enables the most accurate physical-core and RAM
# detection in `benchmark` and thread auto-detection. Without it, the tool
# falls back to /proc parsing and os.cpu_count() heuristics.
psutil
